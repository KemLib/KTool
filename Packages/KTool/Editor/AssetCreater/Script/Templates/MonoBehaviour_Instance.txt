using UnityEngine;

namespace [Namespace]
{
    [AccessModifier] class [NameScript] : MonoBehaviour
    {
        #region Fields

        #endregion
        
        #region Properties
        public static [NameScript] Instance
        {
            get;
            private set;
        }
        #endregion

        #region Methods Unity
        private void Awake()
        {
            if (Instance == null)
            {
                Instance = this;
                DontDestroyOnLoad(gameObject);
            }
            else
            {
                Destroy(gameObject);
            }
        }
        private void OnDestroy()
        {
            if (Instance != null && Instance.GetEntityId() == GetEntityId())
                Instance = null;
        }
        #endregion

        #region Methods

        #endregion
    }
}
